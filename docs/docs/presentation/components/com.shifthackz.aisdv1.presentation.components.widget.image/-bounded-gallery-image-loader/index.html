BoundedGalleryImageLoader

internal class BoundedGalleryImageLoader(val source: GalleryReadGateway, val decoder: GalleryImageDecoder, val budget: GalleryMemoryBudget, val dispatchers: DispatchersProvider) : GalleryImageLoader(source)
