cmake_minimum_required(VERSION 3.22)
project(SdaiMnnApple LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Upstream selects ARM assembly from ARCHS; CMake leaves its iOS processor unset.
set(ARCHS arm64)
get_filename_component(ADAPTER "${CMAKE_CURRENT_LIST_DIR}/Runtime" ABSOLUTE)
include_directories("${ADAPTER}")
add_compile_definitions(SDAI_MNN_PRIVATE_MMAP=1)
add_compile_options(-fexceptions "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=."
    "-ffile-prefix-map=${CMAKE_BINARY_DIR}=." "-ffile-prefix-map=${MNN_SOURCE}=.")
set(MNN_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MNN_SEP_BUILD OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_LLM ON CACHE BOOL "" FORCE)
set(MNN_BUILD_DIFFUSION ON CACHE BOOL "" FORCE)
set(MNN_BUILD_OPENCV ON CACHE BOOL "" FORCE)
set(MNN_IMGCODECS ON CACHE BOOL "" FORCE)
set(MNN_LOW_MEMORY ON CACHE BOOL "" FORCE)
set(MNN_METAL ON CACHE BOOL "" FORCE)
set(MNN_METAL_TENSOR OFF CACHE BOOL "" FORCE)
set(MNN_OPENCL OFF CACHE BOOL "" FORCE)
set(MNN_OPENMP OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_LLM_OMNI OFF CACHE BOOL "" FORCE)
set(MNN_LLM_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(LLM_SUPPORT_HTTP_RESOURCE OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_FOR_ANDROID_COMMAND OFF CACHE BOOL "" FORCE)
add_subdirectory("${MNN_SOURCE}" mnn EXCLUDE_FROM_ALL)
target_sources(diffusion PRIVATE "${ADAPTER}/sdai_unet_graph.cpp")
target_include_directories(diffusion PRIVATE "${ADAPTER}"
    "${MNN_SOURCE}/tools/cv/include" "${MNN_SOURCE}/schema/current"
    "${MNN_SOURCE}/3rd_party/flatbuffers/include")
target_sources(MNN PRIVATE MnnBridge.mm)
target_include_directories(MNN PRIVATE include "${ADAPTER}" "${MNN_SOURCE}/include"
    "${MNN_SOURCE}/transformers/diffusion/engine/include")
target_link_libraries(MNN PUBLIC "-framework Foundation" "-framework Metal" "-framework CoreGraphics")
